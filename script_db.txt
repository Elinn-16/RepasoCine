USE [master]
GO

IF NOT EXISTS (SELECT name FROM sys.databases WHERE name = N'db_cine')
BEGIN
    CREATE DATABASE [db_cine];
END
GO

USE [db_cine]
GO

IF OBJECT_ID('dbo.Funciones', 'U') IS NOT NULL DROP TABLE dbo.Funciones;
IF OBJECT_ID('dbo.Salas', 'U') IS NOT NULL DROP TABLE dbo.Salas;
GO

CREATE TABLE dbo.Salas (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Nombre NVARCHAR(100) NOT NULL,
    Formato NVARCHAR(10) NOT NULL,
    EstaActiva BIT NOT NULL
);
GO

CREATE TABLE dbo.Funciones (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Pelicula NVARCHAR(150) NOT NULL,
    DuracionMinutos INT NOT NULL,
    Fecha DATE NOT NULL,
    EstaFinalizada BIT NOT NULL,
    IdSala INT NOT NULL,
    CONSTRAINT FK_Funciones_Salas FOREIGN KEY (IdSala) REFERENCES dbo.Salas(Id)
);
GO
